cmake_minimum_required(VERSION 3.10)
project(halo_livox CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LIVOX_SDK2_DIR "" CACHE PATH "Livox-SDK2 source and PIC build")
set(LIVOX_SDK2_LIB "${LIVOX_SDK2_DIR}/build/sdk_core/liblivox_lidar_sdk_static.a")
if(NOT EXISTS "${LIVOX_SDK2_LIB}")
  message(FATAL_ERROR "Build Livox-SDK2 with -DCMAKE_POSITION_INDEPENDENT_CODE=ON first, or set LIVOX_SDK2_DIR")
endif()

find_package(Threads REQUIRED)
add_library(halo_livox SHARED livox_driver.cpp)
target_include_directories(halo_livox PRIVATE ${LIVOX_SDK2_DIR}/include)
target_link_libraries(halo_livox PRIVATE ${LIVOX_SDK2_LIB} Threads::Threads)

option(HALO_LIVOX_BUILD_TESTS "Build native packet contract tests" OFF)
if(HALO_LIVOX_BUILD_TESTS)
  enable_testing()
  add_executable(livox_packet_test tests/test_packets.cpp)
  target_include_directories(livox_packet_test PRIVATE ${LIVOX_SDK2_DIR}/include)
  target_compile_options(livox_packet_test PRIVATE -UNDEBUG)
  target_link_libraries(livox_packet_test PRIVATE ${LIVOX_SDK2_LIB} Threads::Threads)
  add_test(NAME livox_packet_contract COMMAND livox_packet_test)
endif()
